#in feature = 128 * 8 * 8 = 8192
#batch = 64
#out feature = 256

#set constant
.set IF, 2048 # 8192 / 4
.set OF, 256
.set B, 64
.set M, 64
.set K, 36 # 12 * 3
.set N, 128 # 32 * 4
.set outf_offest, 48 # N * 4
.set inf_offest, 96 # K * 3
.set b_offest, 64 # M
.set k_offest, 18 # tk * 3
.set n_offest, 32 # tn * 4
.set m_offest, 1  # mode

.set PE_ARRAY_WEIGHT_SIZE, 576  # 48 * 12
.set PE_ARRAY_IPSUM_SIZE, 32    # 8 * 1 * 4
.set PE_ARRAY_IFMAP_SIZE, 144   # 48 * 3
.set DRAM_IFMAP_SIZE, 8192      # M * K * 12
.set DRAM_WEIGHT_SIZE, 18432    # K * N * 48
.set DRAM_OPSUM_SIZE, 12288     # B * N * 16

SET_ID
CFG_SET DRAM_IFMAP_BASE,     0
CFG_SET DRAM_WEIGHT_BASE,    1
CFG_SET DRAM_OFMAP_BASE,     2
CFG_SET GLB_IFMAP_BASE,      3
CFG_SET GLB_WEIGHT_BASE,     4
CFG_SET GLB_OPSUM_BASE,      5
CFG_SET OF_SIZE, 6
CFG_SET IF_SIZE, 7
CFG_SET B_SIZE,  8
CFG_SET K_SIZE,  9
CFG_SET N_SIZE,  10
CFG_SET M_SIZE,  11
CFG_SET MODE,    12
CFG_SET DATA_FLOW, 13

LOADI outf,  0
LOADI inf,   0
LOADI b,     0
LOADI k,     0
LOADI n,     0
LOADI m,     0
LOADI REG[8], 0
LOADI REG[9], 0
LOADI REG[10], 0
LOADI REG[11], 0
LOADI REG[12], 0
LOADI REG[13], 0
LOADI REG[14], 0

#load IFMAP to GLB
DMA_LOAD_IFMAP   DRAM_IFMAP_BASE,    outf, DRAM_IFMAP_SIZE
WAIT DMA
#load WEIGHT to GLB
DMA_LOAD_WEIGHT  DRAM_WEIGHT_BASE,   outf, DRAM_IFMAP_SIZE
WAIT DMA
#load PSUM to GLB
DMA_LOAD_PSUM    DRAM_OFMAP_BASE,    outf, DRAM_OPSUM_SIZE
WAIT DMA

# Output feature tiles
loop_outf:
    loop_inf:
        loop_b:
            loop_k:
                loop_n:
                    #load weight to pe array
                    CPT_TAGXY WEIGHT
                    #compute index
                    MULI REG[11], k, N
                    ADD REG[11], REG[11], n
                    G2P GLB_WEIGHT_BASE, REG[11], PE_ARRAY_WEIGHT_SIZE
                    WAIT GLB
                    loop_m:
                        #load ipsum to pe array
                        CPT_TAGXY IPSUM
                        #compute index
                        MULI REG[12], b, N
                        MULI REG[13], m, N
                        ADD REG[12], REG[12], n
                        ADD REG[12], REG[12], REG[13]
                        G2P GLB_OPSUM_BASE, REG[12], PE_ARRAY_IPSUM_SIZE
                        WAIT GLB
                        #load ifmap to pe array
                        CPT_TAGXY IFMAP
                        #compute index
                        MULI REG[14], m, K
                        G2P GLB_IFMAP_BASE, REG[14], PE_ARRAY_IFMAP_SIZE
                        WAIT GLB
                        #start pe array
                        COMPUTE
                        WAIT PE_ARRAY
                        #write to GLB
                        CPT_TAGXY OPSUM
                        P2G_OPSUM GLB_OPSUM_BASE, REG[12], PE_ARRAY_IPSUM_SIZE
                        LOOP M_SIZE, m, loop_m, m_offest
                    LOOP N_SIZE, n, loop_n, n_offest
                LOOP K_SIZE, k, loop_k, k_offest
            #load IFMAP to GLB
            #compute index
            ADDI REG[10], b, b_offest
            MULI REG[10], REG[10], IF
            ADDI REG[10], REG[10], inf #REG[10] = ((b + b_offest) * in_feature + inf);
            DMA_LOAD_IFMAP DRAM_IFMAP_BASE, REG[10], DRAM_IFMAP_SIZE
            WAIT DMA
            LOOP B_SIZE, b, loop_b, b_offest
        #load WEIGHT to GLB
        #compute index
        ADDI REG[9], inf, inf_offest
        MULI REG[9], REG[9], OF
        ADDI REG[9], REG[9], outf_offest #REG[9] = ((inf + inf_offest) * out_features + outf);
        DMA_LOAD_WEIGHT DRAM_WEIGHT_BASE, REG[9], DRAM_WEIGHT_SIZE
        WAIT DMA
        LOOP IF_SIZE, inf, loop_inf, inf_offest
    #write opsum to DRAM
    CPT_TAGXY OPSUM
    DMA_STORE_OFMAP DRAM_OFMAP_BASE, outf, DRAM_OPSUM_SIZE
    WAIT DMA
    #load PSUM to GLB
    #compute index
    ADDI REG[8], outf, outf_offest #REG[8] = (outf + outf_offest);
    DMA_LOAD_PSUM DRAM_OFMAP_BASE, REG[8], DRAM_OPSUM_SIZE
    WAIT DMA
    LOOP OF_SIZE, outf, loop_outf, outf_offest
