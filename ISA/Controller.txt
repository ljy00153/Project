# Output feature tiles
for outf in range(0, out_features, N * 4)
    #load PSUM to GLB
    compute_ipsum_index(DRAM_ipsum_idx)
    DMA(DRAM_TO_GLB, DRAM_ipsum_idx, size = B * N * 16) #4byte
    # Input feature tiles
    for inf in range(0, in_features / 4, K * 3)
        compute_weight_index(DRAM_weight_idx)
        DMA(DRAM_TO_GLB, DRAM_weight_idx, size = K * N * 48) #4byte
        # Batch loop
        for b in range(0, batch, M)
            compute_ifmap_index(DRAM_ifmap_idx)
            DMA(DRAM_TO_GLB, DRAM_ifmap_idx, size = M * K * 12)
            for k in range(0, K * 3, tk * 3)
                for n in range(0, N * 4, tn * 4)
                    compute_weight_tag_x(weight_x)
                    compute_weight_tag_x(weight_y)
                    compute_weight_index(weight_idx)
                    GLB_to_PE(weight_idx, weight_x, weight_y, size = 12 * 48) #4byte
                    for m in range(0, M, mode)
                        compute_ipsum_tag_x(ipsum_x)
                        compute_ipsum_tag_y(ipsum_y)
                        compute_ipsum_index(ipsum_idx)
                        GLB_to_PE(ipsum_idx, ipsum_x, ipsum_y, size = tn) #4byte

                        compute_ifmap_tag_x(ifmap_x)
                        compute_ifmap_tag_y(ifmap_y)
                        compute_ifmap_index(ifmap_idx)
                        GLB_to_PE(ifmap_idx, ifmap_x, ifmap_y, size = tk * 3) #4byte

                        # PE_compute()
                        compute_opsum_tag_x(opsum_x)
                        compute_opsum_tag_y(opsum_y)
                        compute_opsum_index(opsum_idx)
                        PE_to_GLB(opsum_idx, opsum_x, opsum_y, size = 4) #4byte
    compute_output_idx(DRAM_opsum_idx)
    DMA(GLB_to_DRAM, DRAM_opsum_idx, size = B * N * 16)
