#Register
    CSR 32bit
        CSR[0]: DRAM_IFMAP_BASE
        CSR[1]: DRAM_WEIGHT_BASE
        CSR[2]: DRAM_OFMAP_BASE

        CSR[3]: GLB_IFMAP_BASE
        CSR[4]: GLB_WEIGHT_BASE
        CSR[5]: GLB_OPSUM_BASE

        CSR[6]: OF_SIZE
        CSR[7]: IF_SIZE
        CSR[8]: B_SIZE
        CSR[9]: K_SIZE
        CSR[10]: N_SIZE
        CSR[11]: M_SIZE

        CSR[12]: MODE
        CSR[13]: DATA_FLOW
    Register 32bit
        REG[0]: 0
        loop:
            REG[1]: outf // max: out_features / N * 4 + 1
            REG[2]: inf  // max: (in_features / 4) / K * 3 + 1
            REG[3]: b    // max: batch / M + 1
            REG[4]: k    // max: K * 3 / tk * 3 + 1
            REG[5]: n    // max: N / tn * 8 + 1
            REG[6]: m    // max: M / mode + 1
        compute:
            REG[7]: valid_e 
        General Register
            REG[8] ~ REG[15]
    TAG 3bit
        TAG[0]: count_ifmap_y
        TAG[1]: count_ifmap_x
        TAG[2]: count_weight_y
        TAG[3]: count_weight_x
        TAG[4]: count_ipsum_y
        TAG[5]: count_ipsum_x
        TAG[6]: count_opsum_y
        TAG[7]: count_opsum_x

            
//00000x
OP_NOP                      = 6'b000000 // no operation
OP_CFG_SET                  = 6'b000001 // SET CSR
OP_SET_ID                   = 6'b000010 // SET_LN、SET_XID、SET_YID

//0001xx
OP_DMA_LOAD_IFMAP           = 6'b000100 // DMA read ifmap
OP_DMA_LOAD_WEIGHT          = 6'b000101 // DMA read weights
OP_DMA_LOAD_PSUM            = 6'b000110 // DMA read psum
OP_DMA_STORE_OFMAP          = 6'b000111 // DMA write ofmap

//00100x
OP_G2P	                    = 6'b001000	// GLB → PE
OP_P2G_OPSUM	            = 6'b001001	// PE → GLB opsum

//0100xx
OP_CPT_INDEX	            = 6'b010000	//compute index in GLB

//011000
OP_CPT_TAGXY	            = 6'b011000 //compute and SET tagX and tagY for BUS
                                        //ifmap:    tagX, tagY
                                        //weight:   tagX, tagY
                                        //ipsum:    tagX, tagY
                                        //opsum:    tagX, tagY
                                        //total 4 kind of tag -> 8 small Register
//0111xx
OP_COMPUTE                  = 6'b011100 // set pe enable
OP_WAIT                     = 6'b011101 // wait for ready of pe
OP_JUMP                     = 6'b011110 // absolute jump (pc = imm)
OP_LOOP                     = 6'b011111 // loop counter += offset, if loop counter > certain value, stop looping

//10000x
OP_LOADI                    = 6'b100000 // rd = imm
OP_ADDI                     = 6'b100001 // rd = rs + imm
OP_MULI                     = 6'b100010 // rd = rs1 * imm (unsigned)

//1001xx
OP_ADD                      = 6'b100100 // rd = rs1 + rs2


//111111
OP_END                      = 6'b111111 // program end

CSR[31:0] [3:0]//size, number
REG[31:0] [3:0]//size, number
TAG[2:0]  [2:0]//size, number

FMT-TAG
31      8 7     6 5       0
+---------+------+--------+
| reserve | type | opcode |
+---------+------+--------+
Syntax: CPT_TAGXY type
    case(type)
        IFMAP   : tagY = 0// count_ifmap_y,     tagX = 1// count_ifmap_x
        WEIGHT  : tagY = 2// count_weight_y,    tagX = 3// count_weight_x
        IPSUM   : tagY = 4// count_ipsum_y,     tagX = 5// count_ipsum_x 
        OPSUM   : tagY = 6// count_opsum_y,     tagX = 7// count_opsum_x 
    output_tagY = TAG[tagY], output_tagX = TAG[tagX] 
    update next count_xxx_y and count_xxx_x
    //ifmap GIN bus tag setting
        ifmap_bus_x: bus_tag = 0
        ifmap_bus_y: bus_tag = i, i from 0 to 5
        count_ifmap_x = 0   
        count_ifmap_y++     
    //weight GIN bus tag setting
        weight_bus_x: bus_tag = i, i from 0 to 7
        weight_bus_y: bus_tag = i, i from 0 to 5
        count_weight_x = (PE_weight_num == 12)? count_weight_x++ ： count_weight_x
        count_weight_y = (count_weight_x == 7)? count_weight_y++ : count_weight_y
    //ipsum GIN bus tag setting
        ipsum_bus_x: bus_tag = i, i from 0 to 7
        ipsum_bus_y: bus_tag = i, i from 0 to 5
        //only PE at ROW[0] will receive ipsum
        count_ipsum_x = (PE_ipsum_num == 4)? count_weight_x++ ： count_weight_x
        count_ipsum_y = 0
    //opsum GIN bus tag setting
        opsum_bus_x: bus_tag = i, i from 0 to 7
        opsum_bus_y: bus_tag = i, i from 0 to 5
        count_opsum_x = (PE_opsum_num == 4)? count_weight_x++ ： count_weight_x
        count_opsum_y = 5
