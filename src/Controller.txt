uint8 IFMAP[M * K * 12]
int8 WEIGHT[K * N * 48]
int32 PSUM[B * N * 16]

# Output feature tiles
for outf in range(0, out_features, N * 4)
    DMA(DRAM_TO_GLB, IFMAP, size = M * K * 12)
    DMA(DRAM_TO_GLB, WEIGHT, size = K * N * 48)
    DMA(DRAM_TO_GLB, PSUM, size = B * N * 16)

    # Input feature tiles
    for inf in range(0, in_features / 4, K * 3)
        DMA(DRAM_TO_GLB, IFMAP, size = M * K * 12)
        DMA(DRAM_TO_GLB, WEIGHT, size = K * N * 48)

        # Batch loop
        for b in range(0, batch, M)
            DMA(DRAM_TO_GLB, IFMAP, size = M * K * 12)

            for k in range(0, K * 3, tk * 3)
                for n in range(0, N, tn * 8)
                    GLB_to_PE(WEIGHT)

                    for m in range(0, M, mode)
                        GLB_to_PE(PSUM)
                        GLB_to_PE(IFMAP)
                        # PE_compute()
                        PE_to_GLB(PSUM)

    DMA(GLB_to_DRAM, PSUM, size = B * N * 16)
