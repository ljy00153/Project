# GLB
uint8 IFMAP[M * K * 12]
int8 WEIGHT[K * N * 48]
int32 PSUM[B * N * 16]

# Output feature tiles
for outf in range(0, out_features, N * 4)
    compute_ifmap_index(DRAM_ifmap_idx)
    DMA(DRAM_TO_GLB, DRAM_ifmap_idx, size = M * K * 12) #4byte

    compute_weight_index(DRAM_weight_idx)
    DMA(DRAM_TO_GLB, DRAM_weight_idx, size = K * N * 48) #4byte

    compute_ipsum_index(DRAM_ipsum_idx)
    DMA(DRAM_TO_GLB, DRAM_ipsum_idx, size = B * N * 16) #4byte

    # Input feature tiles
    for inf in range(0, in_features / 4, K * 3)
        compute_ifmap_index(DRAM_ifmap_idx)
        DMA(DRAM_TO_GLB, DRAM_ifmap_idx, size = M * K * 12) #4byte

        compute_weight_index(DRAM_weight_idx)
        DMA(DRAM_TO_GLB, DRAM_weight_idx, size = K * N * 48) #4byte

        # Batch loop
        for b in range(0, batch, M)
            compute_ifmap_index(DRAM_ifmap_idx)
            DMA(DRAM_TO_GLB, DRAM_ifmap_idx, size = M * K * 12)

            for k in range(0, K * 3, tk * 3)
                for n in range(0, N, tn * 8)
                    compute_pe_tag(weight_tag)
                    compute_weight_index(weight_idx)
                    GLB_to_PE(weight_idx, weight_tag, size = 12 * 48) #4byte

                    for m in range(0, M, mode)
                        compute_pe_tag(ipsum_tag)
                        compute_ipsum_index(ipsum_idx)
                        GLB_to_PE(ipsum_idx, ipsum_tag, size = tn) #4byte

                        compute_pe_tag(ifmap_tag)
                        compute_ifmap_index(ifmap_idx)
                        GLB_to_PE(ifmap_idx, ifmap_tag, size = tk * 3) #4byte

                        # PE_compute()
                        compute_opsum_index(opsum_idx)
                        PE_to_GLB(opsum_idx, opsum, size = 4) #4byte

    DMA(GLB_to_DRAM, IFMAP_start + WEIGHT_start, size = B * N * 16)

#GLB_to_PE is parallel
